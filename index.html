#include #include #include #include // ---------------- WIFI ---------------- const char* ssid = "VKD-C32"; const char* pass = "rekhasakthi2007"; // ---------------- THINGSPEAK ---------------- unsigned long myChannelNumber = 3221984; const char* myWriteAPIKey = "AKT7J4I46XGYM7VY"; const char* talkBackID = "64U4IA5216OLCOX8"; const char* talkBackKey = "AKT7J4I46XGYM7VY"; // ---------------- PINS ---------------- #define DHTPIN D3 #define DHTTYPE DHT11 int ledPin = D0; int buzzer = D5; int mq135Pin = A0; int mq2Pin = D2; DHT dht(DHTPIN, DHTTYPE); WiFiClient client; bool sensorsActive = true; unsigned long lastTime = 0; const unsigned long timerDelay = 180000; // 3 min // ------------------------------------------------- void setup() { Serial.begin(115200); // Enhanced startup UI printSeparator(); Serial.println(" GAS MONITORING SYSTEM"); Serial.println(" ESP8266 + ThingSpeak IoT"); printSeparator(); Serial.println(); pinMode(ledPin, OUTPUT); pinMode(buzzer, OUTPUT); pinMode(mq2Pin, INPUT); dht.begin(); Serial.print("Connecting to WiFi: "); Serial.println(ssid); digitalWrite(ledPin, HIGH); int attempts = 0; while (WiFi.status() != WL_CONNECTED && attempts < 20) { delay(500); Serial.print("."); attempts++; } if (WiFi.status() == WL_CONNECTED) { Serial.println("\n✓ WiFi Connected!"); Serial.print("IP Address: "); Serial.println(WiFi.localIP()); digitalWrite(ledPin, LOW); } else { Serial.println("\n✗ WiFi Connection Failed!"); } ThingSpeak.begin(client); Serial.println("✓ ThingSpeak Initialized"); Serial.println(); printSeparator(); Serial.println("System Ready - Monitoring Started"); printSeparator(); } // ------------------------------------------------- String getTalkBackCommand() { HTTPClient http; WiFiClient clientHttp; String url = "http://api.thingspeak.com/talkbacks/" + String(talkBackID) + "/commands/execute.json?api_key=" + String(talkBackKey); http.begin(clientHttp, url); int httpCode = http.GET(); String payload = (httpCode > 0) ? http.getString() : ""; http.end(); if (payload.indexOf("TURN_OFF") != -1) return "TURN_OFF"; if (payload.indexOf("TURN_ON") != -1) return "TURN_ON"; return ""; } // ------------------------------------------------- void printSeparator() { Serial.println("===================================="); } void printSensorStatus(float temperature, float humidity, int gasPPM, bool gasDetected) { Serial.println(); printSeparator(); Serial.println(" SENSOR READINGS"); printSeparator(); // Temperature Serial.print("Temperature: "); Serial.print(temperature, 1); Serial.print(" °C"); if (temperature >= 40) { Serial.print(" [HIGH]"); } Serial.println(); // Humidity Serial.print("Humidity: "); Serial.print(humidity, 1); Serial.println(" %"); // Gas PPM Serial.print("Gas PPM (MQ135): "); Serial.print(gasPPM); if (gasPPM > 900) { Serial.print(" [DANGER]"); } Serial.println(); // Gas Detection Serial.print("Gas Detection (MQ2): "); if (gasDetected) { Serial.println("⚠ GAS DETECTED!"); } else { Serial.println("✓ Safe"); } printSeparator(); // System Status Serial.print("System Status: "); Serial.println(sensorsActive ? "ACTIVE" : "INACTIVE"); // Buzzer Status Serial.print("Buzzer: "); if (temperature >= 40 || gasPPM > 900 || gasDetected) { Serial.println("⚠ ACTIVE"); } else { Serial.println("✓ Silent"); } printSeparator(); } void loop() { // LED blink if WiFi OK if (WiFi.status() == WL_CONNECTED) { digitalWrite(ledPin, !digitalRead(ledPin)); } else { digitalWrite(ledPin, LOW); Serial.println("⚠ WiFi Disconnected!"); } // Check TalkBack String command = getTalkBackCommand(); if (command == "TURN_OFF") { if (sensorsActive) { sensorsActive = false; Serial.println(); printSeparator(); Serial.println("⚠ SENSORS DEACTIVATED"); printSeparator(); } } if (command == "TURN_ON") { if (!sensorsActive) { sensorsActive = true; Serial.println(); printSeparator(); Serial.println("✓ SENSORS ACTIVATED"); printSeparator(); } } if (sensorsActive) { // -------- READ SENSORS -------- float humidity = dht.readHumidity(); float temperature = dht.readTemperature(); int gasPPM = analogRead(mq135Pin); // MAIN GAS VALUE bool gasDetected = (digitalRead(mq2Pin) == LOW); // -------- ENHANCED UI -------- printSensorStatus(temperature, humidity, gasPPM, gasDetected); // -------- BUZZER LOGIC -------- if (temperature >= 40 || gasPPM > 900 || gasDetected) { digitalWrite(buzzer, HIGH); Serial.println("⚠ WARNING: Buzzer Activated!"); } else { digitalWrite(buzzer, LOW); } // -------- THINGSPEAK UPLOAD -------- if (millis() - lastTime > timerDelay || lastTime == 0) { Serial.println(); Serial.println("Uploading data to ThingSpeak..."); ThingSpeak.setField(1, gasPPM); // GAS ThingSpeak.setField(2, (gasDetected ? 1 : 0)); // FIRE/GAS ThingSpeak.setField(3, temperature); // TEMP ThingSpeak.setField(4, humidity); // HUMIDITY int status = ThingSpeak.writeFields( myChannelNumber, myWriteAPIKey ); if (status == 200) { Serial.println("✓ ThingSpeak Upload Successful"); } else { Serial.print("✗ Upload Failed: "); Serial.println(status); } lastTime = millis(); } } delay(2000); // Increased delay for better readability }
